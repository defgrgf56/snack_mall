## 📖 使用指南

### 方式 1：普通页面使用新架构

**旧代码：**
```javascript
const { api } = require('../../config/api.js')

Page({
  data: { loading: true, product: null },
  
  async onLoad(options) {
    try {
      wx.showLoading({ title: '加载中...' })
      const res = await api.getProductDetail(options.id)
      this.setData({ product: res.data, loading: false })
      wx.hideLoading()
    } catch (error) {
      wx.hideLoading()
      wx.showToast({ title: '加载失败', icon: 'none' })
    }
  }
})
```

**新代码：**
```javascript
import { createPage } from '../../mixins/page.mixin'
import api from '../../services/index'

createPage({
  data: { product: null },
  
  async onLoad(options) {
    await this.safeAsync(
      async () => {
        this.setLoading()
        const product = await api.product.getProductDetail(options.id)
        this.setData({ product })
        this.setLoaded()
      },
      { showLoading: true, loadingText: '加载中...' }
    )
  }
})
```

**优势：**
- 代码量减少 40%
- 自动处理 loading 和错误
- 统一的状态管理

### 方式 2：列表页面使用新架构

**旧代码：**
```javascript
Page({
  data: { list: [], page: 1, hasMore: true },
  
  async loadList() {
    if (!this.data.hasMore) return
    const res = await api.getProducts({ page: this.data.page })
    this.setData({
      list: [...this.data.list, ...res.data.items],
      page: this.data.page + 1,
      hasMore: res.data.items.length >= 10
    })
  },
  
  onLoad() { this.loadList() },
  onReachBottom() { this.loadList() },
  onPullDownRefresh() {
    this.setData({ list: [], page: 1, hasMore: true })
    this.loadList().then(() => wx.stopPullDownRefresh())
  }
})
```

**新代码：**
```javascript
import { createListPage } from '../../mixins/page.mixin'
import api from '../../services/index'

createListPage({
  async loadListData(page, pageSize) {
    return await api.product.getProducts({ page, pageSize })
  }
})
```

**优势：**
- 自动处理分页、下拉刷新、触底加载
- 自动处理 empty/error 状态
- 代码量减少 70%

### 方式 3：使用 Store 管理状态

**旧代码：**
```javascript
const app = getApp()

// 设置
app.globalData.token = 'xxx'
wx.setStorageSync('token', 'xxx')

// 获取
const token = app.globalData.token || wx.getStorageSync('token')
```

**新代码：**
```javascript
import store from '../../store/index'

// 设置（自动持久化）
store.setToken('xxx')

// 获取

// 订阅变化
const unsubscribe = store.subscribe('token', (newToken, oldToken) => {
  console.log('Token 变化:', oldToken, '->', newToken)
})

// 取消订阅
unsubscribe()
```

## 🔄 迁移步骤

### 第 1 步：迁移简单页面（如商品详情）

1. 将 `const { api } = require(...)` 改为 `import api from '../../services/index'`
2. 将 `Page({})` 改为 `createPage({})`
3. 使用 `this.safeAsync()` 包装异步操作
4. 使用 `this.setLoading()` / `this.setLoaded()` 管理状态

### 第 2 步：迁移列表页面（如商品列表、订单列表）

1. 使用 `createListPage({})` 替代 `Page({})`
2. 实现 `loadListData(page, pageSize)` 方法
3. 删除手动分页逻辑
4. 删除下拉刷新和触底加载代码（自动处理）

### 第 3 步：迁移用到 globalData 的代码

1. 将 `getApp().globalData.xxx` 改为 `store.getXxx()`
2. 将 `getApp().globalData.xxx = value` 改为 `store.setXxx(value)`

### 第 4 步：替换魔法值

1. 将状态码 `1, 2, 3` 改为 `ORDER_STATUS.PENDING` 等常量
2. 将颜色 `'#FF6B00'` 改为 `THEME.PRIMARY_COLOR`

## 📝 示例：完整重写一个页面

### 重写 `pages/product-detail/product-detail.js`

```javascript
import { createPage } from '../../mixins/page.mixin'
import api from '../../services/index'

createPage({
  data: {
    productId: null,
    product: null,
    quantity: 1,
    isFavorited: false,
  },

  async onLoad(options) {
    if (!options.id) {
      this.showError('商品不存在')
      setTimeout(() => wx.navigateBack(), 1500)
      return
    }

    this.setData({ productId: options.id })
    
    // 并行加载商品详情和收藏状态
    await Promise.all([
      this.loadProduct(),
      this.checkFavorite()
    ])
  },

  onShow() {
    // 更新购物车数量
    getApp().updateCartCount()
  },

  /**
   * 加载商品详情
   */
  async loadProduct() {
    await this.safeAsync(async () => {
      this.setLoading()
      
      const product = await api.product.getProductDetail(this.data.productId)
      
      // 处理图片
      let images = []
      if (product.images) {
        images = typeof product.images === 'string' 
          ? JSON.parse(product.images) 
          : product.images
      }
      if (images.length === 0 && product.cover) {
        images = [product.cover]
      }
      
      this.setData({ product: { ...product, images } })
      this.setLoaded()
    })
  },

  /**
   * 检查收藏状态
   */
  async checkFavorite() {
    if (!this.checkLogin()) return
    
    try {
      const result = await api.http.get(`/favorites/check/${this.data.productId}`)
      this.setData({ isFavorited: result.is_favorited || false })
    } catch (error) {
      console.error('检查收藏状态失败:', error)
    }
  },

  /**
   * 数量变化
   */
  handleMinus() {
    const { quantity } = this.data
    if (quantity > 1) {
      this.setData({ quantity: quantity - 1 })
    }
  },

  handlePlus() {
    const { product, quantity } = this.data
    if (quantity >= product.stock) {
      this.showError('库存不足')
      return
    }
    this.setData({ quantity: quantity + 1 })
  },

  /**
   * 加入购物车
   */
  async handleAddToCart() {
    if (!this.requireLogin()) return
    
    await this.safeAsync(async () => {
      await api.cart.addToCart({
        product_id: this.data.productId,
        quantity: this.data.quantity
      })
      
      this.showSuccess('已加入购物车')
      this.setData({ quantity: 1 })
      
      // 更新购物车数量
      getApp().updateCartCount()
    })
  },

  /**
   * 立即购买
   */
  async handleBuyNow() {
    if (!this.requireLogin()) return
    
    await this.handleAddToCart()
    wx.switchTab({ url: '/pages/cart/cart' })
  },

  /**
   * 收藏/取消收藏
   */
  async handleFavorite() {
    if (!this.requireLogin()) return

    await this.safeAsync(async () => {
      if (this.data.isFavorited) {
        await api.http.delete(`/favorites/product/${this.data.productId}`)
        this.setData({ isFavorited: false })
        this.showSuccess('取消收藏')
      } else {
        await api.http.post('/favorites', { product_id: this.data.productId })
        this.setData({ isFavorited: true })
        this.showSuccess('收藏成功')
      }
    })
  },

  /**
   * 分享商品
   */
  onShareAppMessage() {
    const { product } = this.data
    return {
      title: product.name,
      path: `/pages/product-detail/product-detail?id=${product.id}`,
      imageUrl: product.cover
    }
  }
})
```

A: 支持。微信小程序从基础库 2.0 开始就支持 ES6 模块。

### Q: 性能会变差吗？
A: 不会。新架构增加了请求去重、状态缓存，性能反而更好。
