### 第 3 步：简化异步操作

**旧代码：**
```javascript
async loadProductDetail() {
  this.setData({ loading: true })
  
  try {
    const res = await api.getProductDetail(this.data.productId)
    
    if (res.code === 200) {
      this.setData({
        product: res.data,
        loading: false
      })
    } else {
      wx.showToast({
        title: res.message || '加载失败',
        icon: 'none'
      })
      this.setData({ loading: false })
    }
  } catch (error) {
    console.error('加载商品详情失败', error)
    wx.showToast({
      title: '加载失败',
      icon: 'none'
    })
    this.setData({ loading: false })
  }
}
```

**新代码：**
```javascript
async loadProductDetail() {
  await this.safeAsync(async () => {
    this.setLoading()
    
    const product = await api.product.getProductDetail(this.data.productId)
    
    this.setData({ product })
    this.setLoaded()
  })
}
```

**代码减少 70%！**

### 第 4 步：简化用户交互

**旧代码：**
```javascript
async handleAddToCart() {
  const app = getApp()
  
  if (!app.globalData.token) {
    wx.showModal({
      title: '提示',
      content: '请先登录',
      confirmText: '去登录',
      success: (res) => {
        if (res.confirm) {
          wx.switchTab({ url: '/pages/user/user' })
        }
      }
    })
    return
  }
  
  try {
    wx.showLoading({ title: '加入购物车...' })
    
    const res = await api.addToCart({
      product_id: this.data.productId,
      quantity: this.data.quantity
    })
    
    wx.hideLoading()
    
    if (res.code === 200) {
      wx.showToast({
        title: '已加入购物车',
        icon: 'success'
      })
      
      app.updateCartCount()
      
      this.setData({ quantity: 1 })
    } else {
      wx.showToast({
        title: res.message || '加入失败',
        icon: 'none'
      })
    }
  } catch (error) {
    wx.hideLoading()
    wx.showToast({
      title: '加入失败',
      icon: 'none'
    })
  }
}
```

**新代码：**
```javascript
async handleAddToCart() {
  if (!this.requireLogin()) return
```

```javascript
      console.error('加载商品详情失败', error)
```

### 新代码（约 60 行）

```javascript
import { createPage } from '../../mixins/page.mixin'
import api from '../../services/index'

createPage({
  data: {
    productId: null,
    product: null,
    quantity: 1,
    isFavorited: false
  },

  async onLoad(options) {
    if (!options.id) {
      this.showError('商品不存在')
      setTimeout(() => wx.navigateBack(), 1500)
      return
    }

    this.setData({ productId: options.id })
    await Promise.all([
      this.loadProductDetail(),
      this.checkFavoriteStatus()
    ])
  },

  async loadProductDetail() {
    await this.safeAsync(async () => {
      this.setLoading()
      
      const product = await api.product.getProductDetail(this.data.productId)
      
      let images = []
      if (product.images) {
        images = typeof product.images === 'string' 
          ? JSON.parse(product.images) 
          : product.images
      }
      if (images.length === 0 && product.cover) {
        images = [product.cover]
      }
      
      this.setData({ product: { ...product, images } })
      this.setLoaded()
    })
  },

  async handleAddToCart() {
    if (!this.requireLogin()) return
    
    await this.safeAsync(async () => {
      await api.cart.addToCart({
        product_id: this.data.productId,
        quantity: this.data.quantity
      })
      
      this.showSuccess('已加入购物车')
      this.setData({ quantity: 1 })
      getApp().updateCartCount()
    })
  }
})
```

## 列表页更简单

**旧代码（约 200 行）：**
```javascript
Page({
  data: { list: [], page: 1, hasMore: true, loading: false },
  
  async loadList() {
    if (!this.data.hasMore || this.data.loading) return
    
    this.setData({ loading: true })
    
    try {
      const res = await api.getOrders({
        page: this.data.page,
        limit: 10
      })
      
      const list = this.data.page === 1 
        ? res.data.items 
        : [...this.data.list, ...res.data.items]
      
      this.setData({
        list,
        page: this.data.page + 1,
        hasMore: res.data.items.length >= 10,
        loading: false
      })
    } catch (error) {
      this.setData({ loading: false })
      wx.showToast({ title: '加载失败', icon: 'none' })
    }
  },
  
  onLoad() { this.loadList() },
  onReachBottom() { this.loadList() },
  onPullDownRefresh() {
    this.setData({ list: [], page: 1, hasMore: true })
    this.loadList().then(() => wx.stopPullDownRefresh())
  }
})
```
