## 🔑 核心配置

### 1. 微信开放平台配置

**需要的信息**:
- `AppID`: 小程序ID（在微信公众平台获取）
- `AppSecret`: 小程序密钥（在微信公众平台获取）

**获取位置**:
1. 登录 https://mp.weixin.qq.com/
2. 进入"开发" → "开发管理" → "开发设置"
3. 复制 AppID 和 AppSecret

### 2. 后端环境变量配置

在 `后端API/.env` 文件中添加：

```env
# 微信小程序配置
WX_APPID=your_appid_here
WX_SECRET=your_app_secret_here

# JWT配置
JWT_SECRET=your_jwt_secret_key_here
JWT_EXPIRES_IN=7d
```

**安全建议**:
- ✅ `JWT_SECRET` 使用随机字符串（至少32位）
- ✅ 不要将 `.env` 文件提交到Git
- ✅ `WX_SECRET` 不要泄露

## 💻 前端实现

### 1. app.js - 全局登录管理

```javascript
// app.js
App({
  globalData: {
    userInfo: null,
    token: null,
    apiBase: 'http://localhost:3000/api',
    cartCount: 0
  },

  onLaunch() {
    console.log('小程序启动')
    
    // 检查登录状态
    this.checkLogin()
  },

  /**
   * 检查登录状态
   */
  checkLogin() {
    const token = wx.getStorageSync('token')
    if (token) {
      this.globalData.token = token
      // 验证token是否有效
      this.validateToken()
    }
  },

  /**
   * 验证token有效性
   */
  async validateToken() {
    try {
      const res = await this.request({
        url: '/user/info',
        method: 'GET'
      })
      
      if (res.code === 200) {
        this.globalData.userInfo = res.data
        // 更新购物车数量
        this.updateCartCount()
      } else {
        // token失效，清除登录信息
        this.logout()
      }
    } catch (error) {
      console.error('验证token失败:', error)
      this.logout()
    }
  },

  /**
   * 微信登录
   */
  login() {
    return new Promise((resolve, reject) => {
      wx.showLoading({ title: '登录中...' })
      
      wx.login({
        success: async (loginRes) => {
          if (loginRes.code) {
            try {
              // 发送code到后端换取token
              const response = await this.request({
                url: '/auth/login',
                method: 'POST',
                data: {
                  code: loginRes.code
                }
              })
              
              wx.hideLoading()
              
              if (response.code === 200) {
                const { token, userInfo } = response.data
                
                // 保存登录信息
                this.globalData.token = token
                this.globalData.userInfo = userInfo
                wx.setStorageSync('token', token)
                wx.setStorageSync('userInfo', userInfo)
                
                // 更新购物车数量
                this.updateCartCount()
                
                wx.showToast({
                  title: '登录成功',
                  icon: 'success'
                })
                
                resolve(userInfo)
              } else {
                wx.showToast({
                  title: response.message || '登录失败',
                  icon: 'none'
                })
                reject(response.message)
              }
            } catch (error) {
              wx.hideLoading()
              wx.showToast({
                title: '登录失败，请重试',
                icon: 'none'
              })
              reject(error)
            }
          } else {
            wx.hideLoading()
            reject('获取code失败')
          }
        },
        fail: (error) => {
          wx.hideLoading()
          reject(error)
        }
      })
    })
  },

  /**
   * 退出登录
   */
  logout() {
    this.globalData.token = null
    this.globalData.userInfo = null
    this.globalData.cartCount = 0
    wx.removeStorageSync('token')
    wx.removeStorageSync('userInfo')
    wx.removeTabBarBadge({ index: 2 })
  },

  /**
   * 统一请求方法（带token）
   */
  request({ url, method = 'GET', data = {} }) {
    return new Promise((resolve, reject) => {
      wx.request({
        url: `${this.globalData.apiBase}${url}`,
        method,
        data,
        header: {
          'Content-Type': 'application/json',
          'Authorization': this.globalData.token ? `Bearer ${this.globalData.token}` : ''
        },
        success: (res) => {
          // 401 表示未登录或token失效
          if (res.data.code === 401) {
            this.logout()
            wx.showToast({
              title: '请先登录',
              icon: 'none'
            })
            reject({ code: 401, message: '未登录' })
            return
          }
          
          resolve(res.data)
        },
        fail: (error) => {
          reject(error)
        }
      })
    })
  },

  /**
   * 更新购物车数量
   */
  async updateCartCount() {
    if (!this.globalData.token) {
      return
    }

    try {
      const res = await this.request({
        url: '/cart/count',
        method: 'GET'
      })
      
      if (res.code === 200) {
        this.globalData.cartCount = res.data.count || 0
        
        if (this.globalData.cartCount > 0) {
          wx.setTabBarBadge({
            index: 2,
            text: String(this.globalData.cartCount)
          })
        } else {
          wx.removeTabBarBadge({ index: 2 })
        }
      }
    } catch (error) {
      console.error('更新购物车数量失败:', error)
    }
  }
})
```

### 2. utils/request.js - 请求工具（优化版）

```javascript
// utils/request.js
const app = getApp()

/**
 * 统一请求封装
 */
function request(url, options = {}) {
  const {
    method = 'GET',
    data = {},
    needAuth = true // 是否需要token
  } = options

  return new Promise((resolve, reject) => {
    const header = {
      'Content-Type': 'application/json'
    }

    // 如果需要token，添加Authorization头
    if (needAuth && app.globalData.token) {
      header['Authorization'] = `Bearer ${app.globalData.token}`
    }

    console.log(`发起请求: ${method} ${app.globalData.apiBase}${url}`)

    wx.request({
      url: `${app.globalData.apiBase}${url}`,
      method,
      data,
      header,
      success: (res) => {
        console.log(`请求成功: ${url}`, res.statusCode, res.data)

        // 401 未登录
        if (res.data.code === 401) {
          app.logout()
          
          wx.showModal({
            title: '提示',
            content: '您还未登录，请先登录',
            confirmText: '去登录',
            success: (modalRes) => {
              if (modalRes.confirm) {
                wx.switchTab({
                  url: '/pages/user/user'
                })
              }
            }
          })
          
          reject({ code: 401, message: '未登录' })
          return
        }

        // 返回数据
        if (res.data.code === 200) {
          resolve(res.data.data)
        } else {
          wx.showToast({
            title: res.data.message || '请求失败',
            icon: 'none'
          })
          reject(res.data)
        }
      },
      fail: (error) => {
        console.error(`请求失败: ${url}`, error)
        
        wx.showToast({
          title: '网络请求失败',
          icon: 'none'
        })
        
        reject(error)
      }
    })
  })
}

// 导出便捷方法
module.exports = {
  get: (url, data = {}, needAuth = true) => {
    return request(url, { method: 'GET', data, needAuth })
  },
  
  post: (url, data = {}, needAuth = true) => {
    return request(url, { method: 'POST', data, needAuth })
  },
  
  put: (url, data = {}, needAuth = true) => {
    return request(url, { method: 'PUT', data, needAuth })
  },
  
  delete: (url, data = {}, needAuth = true) => {
    return request(url, { method: 'DELETE', data, needAuth })
  }
}
```

```javascript
// pages/user/user.js
const api = require('../../utils/request')
const app = getApp()

Page({
  data: {
    userInfo: null,
    isLogin: false,
    orderStats: {
      pending: 0,    // 待付款
      paid: 0,       // 待发货
      shipped: 0,    // 待收货

    }

  onLoad() {
    this.checkLoginStatus()
  },

  onShow() {
    this.checkLoginStatus()
  },

  /**
   * 检查登录状态
   */
  checkLoginStatus() {
    const userInfo = app.globalData.userInfo
    
    if (userInfo) {
      this.setData({
        userInfo,
        isLogin: true
      })
      // 加载订单统计
      this.loadOrderStats()
    } else {
      this.setData({
        userInfo: null,
        isLogin: false
      })
    }
  },

  /**
   * 登录
   */
  async onLogin() {
    try {
      // 获取用户授权
      const authRes = await wx.getUserProfile({
        desc: '用于完善用户资料'
      })
      
      // 调用微信登录
      const userInfo = await app.login()
      
      // 更新用户资料（昵称、头像）
      if (authRes.userInfo) {
        await this.updateUserProfile(authRes.userInfo)
      }
      
      // 刷新页面
      this.checkLoginStatus()
      
    } catch (error) {
      console.error('登录失败:', error)
      if (error.errMsg && error.errMsg.includes('getUserProfile:fail auth deny')) {
        wx.showToast({
          title: '您取消了授权',
          icon: 'none'
        })
      }
    }
  },

  /**
   * 更新用户资料
   */
  async updateUserProfile(profile) {
    try {
      await api.post('/auth/update-profile', {
        nickname: profile.nickName,
        avatar: profile.avatarUrl,
        gender: profile.gender
      })
      
      // 更新本地用户信息
      app.globalData.userInfo.nickname = profile.nickName
      app.globalData.userInfo.avatar = profile.avatarUrl
      
    } catch (error) {
      console.error('更新用户资料失败:', error)
    }
  },

  /**
   * 退出登录
   */
  onLogout() {
    wx.showModal({
      title: '提示',
      content: '确定要退出登录吗？',
      success: (res) => {
        if (res.confirm) {
          app.logout()
          
          this.setData({
            userInfo: null,
            isLogin: false,
            orderStats: {
              pending: 0,
              paid: 0,
              shipped: 0,
              uncommented: 0
            }
          })
          
          wx.showToast({
            title: '已退出登录',
            icon: 'success'
          })
        }
      }
    })
  },

  /**
   * 加载订单统计
   */
  async loadOrderStats() {
    try {
      const stats = await api.get('/orders/stats')
      
      this.setData({
        orderStats: {
          pending: stats.pending || 0,
          paid: stats.paid || 0,
          shipped: stats.shipped || 0,
          uncommented: stats.uncommented || 0
        }
      })
    } catch (error) {
      console.error('加载订单统计失败:', error)
    }
  },

  /**
   * 查看订单
   */

      wx.showToast({
        title: '请先登录',
        icon: 'none'
      })
      return
    }
    
    const status = e.currentTarget.dataset.status
    wx.navigateTo({
      url: `/pages/order-list/order-list?status=${status}`
    })
  },

  /**
   * 页面导航（需要登录）
   */
  onNavigate(e) {
    if (!this.data.isLogin) {
      wx.showToast({
        title: '请先登录',
        icon: 'none'
      })
      return
    }
    
    const url = e.currentTarget.dataset.url
    wx.navigateTo({ url })
  }
})
```

## 🔧 后端实现

### 1. 验证中间件 - middleware/auth.js

```javascript
// src/middleware/auth.js
const jwt = require('jsonwebtoken');
const { User } = require('../models');

/**
 * JWT验证中间件
 */
async function authenticateToken(req, res, next) {
  try {
    const authHeader = req.headers['authorization'];
    const token = authHeader && authHeader.split(' ')[1]; // Bearer TOKEN
    
    if (!token) {
      return res.json({
        code: 401,
        message: '未登录',
        data: null
      });
    }
    
    // 验证token
    const decoded = jwt.verify(token, process.env.JWT_SECRET);
    
    // 查找用户
    const user = await User.findByPk(decoded.userId);
    
    if (!user || user.status !== 1) {
      return res.json({
        code: 401,
        message: '用户不存在或已被禁用',
        data: null
      });
    }
    
    // 将用户信息添加到请求对象
    req.user = user;
    req.userId = user.id;
    
    next();
  } catch (error) {
    if (error.name === 'JsonWebTokenError') {
      return res.json({
        code: 401,
        message: 'Token无效',
        data: null
      });
    }
    
    if (error.name === 'TokenExpiredError') {
      return res.json({
        code: 401,
        message: 'Token已过期',
        data: null
      });
    }
    
    console.error('Token验证失败:', error);
    return res.json({
      code: 500,
      message: '服务器错误',
      data: null
    });
  }
}

module.exports = { authenticateToken };
```

## ⚠️ 常见问题

### 问题1: code2session失败

**症状**: 登录时提示"微信登录失败"

**原因**:
- AppID或AppSecret配置错误
- code已失效（5分钟有效期）
- 网络问题

**解决**:
1. 检查`.env`配置
2. 确认AppID和AppSecret正确
3. 重新获取code

### 问题2: Token验证失败

**症状**: 提示"Token无效"或"Token已过期"

**原因**:
- Token格式错误
- JWT_SECRET不匹配
- Token已过期

**解决**:
1. 检查Authorization头格式：`Bearer <token>`
2. 确认JWT_SECRET一致
3. 重新登录获取新token

### 问题3: 401未登录

**症状**: 所有需要登录的接口都返回401

**原因**:
- Token未保存
- Token未携带到请求头
- 中间件未正确配置

**解决**:
1. 检查token是否保存：`wx.getStorageSync('token')`
2. 检查请求头是否携带token
3. 检查后端路由是否使用了`authenticateToken`中间件

## 🔒 安全建议

### 1. 敏感信息保护
- ✅ `.env`文件不要提交到Git
- ✅ `WX_SECRET`不要泄露
- ✅ `JWT_SECRET`使用强随机字符串

### 2. Token管理
- ✅ Token设置合理过期时间（7天）
- ✅ 重要操作需要二次验证
- ✅ 退出登录时清除本地token
